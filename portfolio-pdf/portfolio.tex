    \parbox[t]{\wa pt}{\centering\small\color{sub}#2} & \parbox[t]{\wb pt}{\centering\small\color{sub}#4}

\projectthree{1}
  {UAV Pitch Control and RF Signal Restoration}
  {Linear Systems \textbar{} Jan. 2026 to May 2026}
  {\challenge{A UAV that shakes while collecting RF signals blurs them, so it has to stay level before the signals can be trusted.}
   \approach{I proved with Routh-Hurwitz analysis that P and PI controllers can never stabilize it, then searched 15,625 PID gain combinations and refined the grid 10 times. I restored a blurred RF signal (radar, chirp and communications components at 40dB SNR) with a generalized Wiener filter in MATLAB, and linked the two parts: the controller's settling time sets how long the blur lasts.}
   \outcome{A 52ms rise time, 2.6\% overshoot and 88.3$^\circ$ phase margin, 77\% less error than a direct inverse, and proof that a controller settling in 2s instead of 0.5s makes recovery 1.5$\times$ worse.}\label{p1}}

\projectthree{2}
  {CTGAN Design Discovery for RRAM}
  {Tufts Emerging Circuits and Systems Lab \textbar{} Sept. 2025 to April 2026}
  {\challenge{The published framework I extended used a CVAE, which averages its training data and smooths away the extreme designs where the best performance is found.}
   \approach{I replaced it with a CTGAN that searches those extremes more aggressively and used the framework's physics model (PINN) as a referee to reject impossible designs. The CTGAN collapsed at 500 training samples and stayed noisy at 2,500, so I grew the data to 5,000, where it converged across 3 oxides (Al\textsubscript{2}O\textsubscript{3}, HfO\textsubscript{2} and TiO\textsubscript{2}).}
   \outcome{An Al\textsubscript{2}O\textsubscript{3} design with 4.84$\times$10\textsuperscript{9} cycles against the published 3.85$\times$10\textsuperscript{9} (25\% higher endurance) and 310.7pJ against 350.6pJ (11\% less energy), with slower switching (6.57ns vs 4.07ns).}\label{p2}}
  {rram-baseline.png}{Published CVAE results}
  {rram-pareto.png}{My CTGAN with 5,000 samples}
  {rram-evolution.png}{The CTGAN with 500 samples (left) and 2,500 (right)}

\projectfour{3}
  {DJ Board on an STM32\\[-2pt]{\fontsize{15}{19}\selectfont\mdseries\color{sub}(Audio Recorder and Effects Processor)}}
  {Embedded Systems \textbar{} Jan. 2026 to May 2026}
  {\challenge{Seven real-time audio effects had to fit in the STM32's 64KB of on-chip memory, and the first playback came out with a high-pitched hiss and clicks.}
   {\small\color{sub}Microphone $\rightarrow$ ADC $\rightarrow$ buffer $\rightarrow$ effects $\rightarrow$ DAC $\rightarrow$ amplifier $\rightarrow$ speaker}\par\vspace{9pt}
   \approach{To remove the hiss, I used a 4kHz timer interrupt to record and play every sample at the same rate, and lengthened the ADC sampling time to 12.5 cycles so the microphone signal settles before each reading. The clicks came from sudden jumps in the speaker signal, so I held the output at mid-rail between clips and faded each clip in and out over 256 samples (64ms). Dropping from 8kHz to 4kHz stretched each recording from 2 to 5 seconds.}
   \outcome{5 seconds of 12-bit audio in a 40KB buffer, all 7 effects (echo, reverse, speed, loop size, pitch up, pitch down, volume), and a playback waveform that matched the original on the oscilloscope.}\label{p3}}

\projectfour{4}
  {FPGA Audio Synthesizer}
  {Digital Logic Circuits \textbar{} Sept. 2025 to Dec. 2025}
  {\challenge{Each octave doubles a note's frequency, and my first design calculated every note by dividing on the chip, which needed over 250\% of the FPGA's space. Chords of more than 2 notes, or in low octaves, turned into static.}
   \approach{A tuner showed every note playing an octave low, which meant the clock ran at twice the rate in the code. Since dividing by 2 is the same as shifting bits, I hand-calculated one octave, used bit shifts for the rest, and widened the period counter so notes below 92Hz could play. To find the static, I plotted C and E summed in octave 3 and octave 7 in Desmos, then mixed the notes with PWM on a 3-bit bus, the best balance of volume and clarity.}
   \outcome{All 12 notes across 8 octaves (16Hz to 8kHz), chords of 5 to 6 notes in the lowest octave, and every note at once in the highest.}\label{p4}}
