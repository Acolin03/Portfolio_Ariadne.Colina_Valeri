    }

\project
  {DJ Board on an STM32}
  {EE 14 Embedded Systems \textbar{} Jan. 2026 to May 2026}
  {STM32L432KC \textbar{} C \textbar{} ADC \& DAC \textbar{} Oscilloscope}
  {dj-circuit.png}{The board: microphone, speaker and effect buttons}
  {dj-scope.png}{Oscilloscope captures of the effect outputs}
  {
    \item A real-time audio effects processor records sound and remixes it through 7 effects.
    \item Every effect depends on a clean recording.
    \item The initial playback was marred by a high-pitched hiss and audible clicks.
  }
  {
    \solution{
      \item Implemented a timer-driven state machine that records and plays every sample at a constant rate.
      \item Allowed each microphone reading additional time to complete so that no sample was cut short.
      \item Held the output at a steady resting level between clips and faded each clip in and out.
    }
    \result{
      \item Oscilloscope measurements confirmed that playback matched the original recording.
      \item All 7 effects operate within 64KB of on-chip memory.
    }
  }

\project
  {FPGA Audio Synthesizer}
  {ES 4 Digital Logic Circuits \textbar{} Sept. 2025 to Dec. 2025}
  {SystemVerilog \textbar{} FPGA \textbar{} PWM}
  {fpga-board.png}{The complete board}
  {fpga-octaves.png}{The switch that selects the octave}
  {
    \item Each octave doubles a note's frequency.
    \item The initial design computed every note through division on the chip.
    \item Division required more than 250\% of the FPGA's capacity.
  }
  {
    \solution{
      \item Hand-calculated one octave and derived the rest through bit shifts, since dividing by 2 is equivalent to a bit shift.
      \item Mixed the notes with PWM after testing revealed that chords distorted.
    }
    \result{
      \item An 8-octave synthesizer, from 16Hz to 8kHz, for a MIDI keyboard.
      \item Chords of up to 6 notes play clearly.
    }
  }

\project
  {CTGAN Design Discovery for RRAM}
  {Tufts Emerging Circuits and Systems Lab \textbar{} Sept. 2025 to April 2026}
  {Python \textbar{} PyTorch \textbar{} pandas \textbar{} CUDA \textbar{} CTGAN \textbar{} PINN}
  {rram-pareto.png}{Final results with 5,000 training samples}
  {rram-evolution.png}{The same search with 500 samples (left) and 2,500 (right)}
  {
    \item A published AI framework designs RRAM memory devices.
    \item Its model averaged the training data, smoothing away the extreme designs where peak performance lies.
  }
  {
    \solution{
      \item Replaced the model with a CTGAN, which explores those extremes more aggressively.
      \item Scaled the training data tenfold, to 5,000 samples, so that the CTGAN could learn reliably.
      \item Employed the framework's physics model as a referee to reject impossible designs.
    }
    \result{
      \item 25\% higher endurance than the published result.
      \item 11\% lower switching energy.
      \item Slightly slower switching: 6.57ns versus 4.07ns.
    }
  }

\project
  {UAV Pitch Control and RF Signal Restoration}
  {EE 23 Linear Systems \textbar{} Jan. 2026 to May 2026}
  {MATLAB \textbar{} PID control \textbar{} Wiener filtering}
  {uav-controllers.png}{P, PI, PD and PID controllers on the same UAV model}
  {uav-wiener.png}{Wiener filter error (left) and the restored RF signal (right)}
  {
    \item Stabilizing a UAV in pitch is the first step toward collecting clear RF signals from the air.
    \item Even a stable UAV shakes slightly and blurs the signals it receives.
  }
  {
    \solution{
      \item Demonstrated through Routh-Hurwitz analysis that P and PI controllers cannot stabilize the UAV.
      \item Searched 15,625 gain combinations for the optimal PID settings.
      \item Restored a blurred, noisy RF signal with a Wiener filter.
    }
    \result{
      \item 52ms rise time, 2.6\% overshoot and 500ms settling time, within real UAV specifications.
      \item 77\% lower error than a direct inverse (MSE 2.272 to 0.521).
    }
  }
