\newcommand{\maxpic}{250}  % tallest a picture may be, in points (72 points = 1 inch)
\newcommand{\twopics}[4]{%
  \sbox0{\includegraphics[height=100pt]{#1}}\picA=\wd0
  \sbox2{\includegraphics[height=100pt]{#3}}\picB=\wd2
  \edef\pich{\fpeval{min(\maxpic, (\linewidth - \picgap) * 100 / (\picA + \picB))}}%
  \edef\wa{\fpeval{\pich * \picA / 100}}\edef\wb{\fpeval{\pich * \picB / 100}}%
  \begin{center}
  \begin{tabular}{@{}c@{\hspace{\picgap}}c@{}}
    \includegraphics[height=\pich pt]{#1} & \includegraphics[height=\pich pt]{#3} \\[3pt]
    \parbox[t]{\wa pt}{\centering\small\color{sub}#2} & \parbox[t]{\wb pt}{\centering\small\color{sub}#4}
  \end{tabular}
  \end{center}}

% Two wide pictures stacked, each with a caption, fitted inside a #5 by #6 inch box.

  \sbox2{\includegraphics[height=100pt]{#3}}\picB=\wd2
  \edef\picw{\fpeval{min(#5 * 72.27, (#6 * 72.27 - 60) / (100 / \picA + 100 / \picB))}}%

  \includegraphics[width=\picw pt]{#1}\par\vspace{2pt}
  \parbox{\picw pt}{\centering\small\color{sub}#2}\par\vspace{12pt}
  \includegraphics[width=\picw pt]{#3}\par\vspace{2pt}
  \parbox{\picw pt}{\centering\small\color{sub}#4}\par}

\projectwide{1}
  {UAV Pitch Control and RF Signal Restoration}
  {Linear Systems \textbar{} Jan. 2026 to May 2026}
  {\challenge{A UAV that shakes while collecting RF signals blurs them, so it has to stay level first.}
   \approach{I proved with Routh-Hurwitz analysis that P and PI controllers can never stabilize it, searched 15,625 gain combinations for the best PID settings, then restored a blurred, noisy RF signal with a Wiener filter in MATLAB.}
   \outcome{A 52ms rise time and 2.6\% overshoot, within real UAV specs, and 77\% less error than a direct inverse.}\label{p1}}
  {uav-controllers.png}{P, PI, PD and PID controllers on the same UAV model}
  {uav-wiener.png}{Wiener filter error (left) and the restored RF signal (right)}

\projectwide{2}
  {CTGAN Design Discovery for RRAM}
  {Tufts Emerging Circuits and Systems Lab \textbar{} Sept. 2025 to April 2026}
  {\challenge{The published framework I extended used a CVAE, which averages its training data and smooths away the extreme designs where the best performance is found.}
   \approach{I replaced it with a CTGAN that searches those extremes more aggressively, used the framework's physics model (PINN) to reject impossible designs, and grew the training data tenfold to 5,000 samples after the model collapsed at 500.}
   \outcome{A design with 25\% higher endurance and 11\% lower switching energy than the published result, with slightly slower switching (6.57ns vs 4.07ns).}\label{p2}}
  {rram-pareto.png}{Final results with 5,000 training samples}
  {rram-evolution.png}{The same search with 500 samples (left) and 2,500 (right)}

\projectfour{3}
  {DJ Board on an STM32}
  {Embedded Systems \textbar{} Jan. 2026 to May 2026}
  {\challenge{Seven real-time audio effects had to fit in the STM32's 64KB of on-chip memory, and the first playback came out with a high-pitched hiss and clicks.}
   {\small\color{sub}Microphone $\rightarrow$ ADC $\rightarrow$ buffer $\rightarrow$ effects $\rightarrow$ DAC $\rightarrow$ amplifier $\rightarrow$ speaker}\par\vspace{9pt}
   \approach{To remove the hiss, I wrote a timer-driven state machine so every sample was recorded and played at the same steady rate, and gave each ADC reading time to finish so no sample was cut short. The clicks came from sudden jumps in the speaker signal, so I held the output at a steady resting level between clips and faded each clip in and out.}
   \outcome{All 7 effects (echo, reverse, speed, loop size, pitch up, pitch down, volume) run within 64KB, and the playback waveform on the oscilloscope matched the original recording.}\label{p3}}
  {dj-circuit.png}{The board}
  {dj-hardware.png}{Hardware: wiring diagram}
  {dj-software.png}{Software: the state machine}
  {dj-scope.png}{Oscilloscope captures of the effects}

\projectside{4}
  {FPGA Audio Synthesizer}
  {Digital Logic Circuits \textbar{} Sept. 2025 to Dec. 2025}
  {\challenge{Each octave doubles a note's frequency, and my first design calculated every note by dividing on the chip, which needed over 250\% of the FPGA's space.}
   \approach{Since dividing by 2 is the same as shifting bits, I hand-calculated one octave and used bit shifts for the rest, then mixed the notes with PWM after chords distorted.}
   \outcome{A full 12-note scale across 8 octaves (16Hz to 8kHz) on a MIDI keyboard, with chords of up to 6 clear notes.}\label{p4}}
  {fpga-board.png}{The complete board}
  {fpga-octaves.png}{The switch that selects the octave}
